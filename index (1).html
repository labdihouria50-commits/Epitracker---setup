4. Lock it with the final rules

After steps 2 and 3: Firebase console → Firestore Database → Rules → select all, paste, Publish. Statistics stay public, the patient registry and users are for logged-in staff only.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function isAdmin() { return signedIn() && role() == 'admin'; }
    function isStaff() { return signedIn() && (role() == 'admin' || role() == 'gerant'); }

    match /{coll}/{id} {
      allow read: if coll in ['table_01', 'table_02', 'table_03', 'table_04', 'config', 'schemas'];
      allow write: if isStaff() && coll in ['table_01', 'table_02', 'table_03', 'table_04'];
    }
    match /tables_meta/{id} { allow read: if true; allow write: if isAdmin(); }
    match /config/{id}  { allow write: if isAdmin(); }
    match /schemas/{id} { allow write: if isAdmin(); }
    match /table_05/{id} { allow read, write: if isStaff(); }
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }
  }
}
